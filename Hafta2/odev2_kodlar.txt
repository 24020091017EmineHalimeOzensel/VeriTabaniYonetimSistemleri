CREATE DATABASE IF NOT EXISTS universite_db;
USE universite_db;

CREATE TABLE IF NOT EXISTS dersler (
    id INT AUTO_INCREMENT PRIMARY KEY,
    ders_adi VARCHAR(100)
);

CREATE TABLE IF NOT EXISTS ogrenciler (
    id INT AUTO_INCREMENT PRIMARY KEY,
    ad_soyad VARCHAR(50),
    sehir VARCHAR(50),
    vize_notu INT,
    final_notu INT,
    ders_id INT
);
